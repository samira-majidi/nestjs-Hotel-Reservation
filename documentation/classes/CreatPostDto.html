import {
  IsArray,
  IsDate,
  IsEnum,
  IsNotEmpty,
  IsOptional,
  IsString,
  IsUrl,
  Matches,
  MaxLength,
  MinLength,
  ValidateNested,
} from 'class-validator';
import { PostType } from '../enums/postType.enum';
import { Status } from '../enums/status.enum';
import { CreatPostMetaOptionDto } from './creat-post-meta-option.dto';
import { Type } from 'class-transformer';
import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
export class CreatPostDto {
  @ApiProperty()
  @IsString()
  @IsNotEmpty()
  @MinLength(3)
  title: string;

  @ApiProperty()
  @IsEnum(PostType)
  @IsNotEmpty()
  postType: PostType;

  @ApiProperty()
  @IsString()
  @IsNotEmpty({ message: 'Slug الزامی است' })
  @MinLength(3, { message: 'Slug باید حداقل ۳ کاراکتر باشد' })
  @MaxLength(100, { message: 'Slug حداکثر ۱۰۰ کاراکتر می‌تواند باشد' })
  @Matches(/^[a-z0-9]+(?:-[a-z0-9]+)*$/, {
    message:
      'Slug فقط حروف کوچک، عدد و خط تیره (kebab-case) مجاز است — بدون شروع/پایان با خط تیره',
  })
  slug: string;

  @ApiProperty()
  @IsEnum(Status)
  @IsNotEmpty()
  status: Status;

  @ApiPropertyOptional()
  @IsString()
  @IsOptional()
  content?: string;

  @ApiPropertyOptional()
  @IsOptional()
  schema?: Record<string, any>;

  @ApiPropertyOptional()
  @IsOptional()
  @IsUrl()
  featuredImageUrl?: string;

  @ApiPropertyOptional()
  @IsOptional()
  @IsDate()
  @Type(() => Date)
  publishedOn?: Date;

  @ApiPropertyOptional()
  @IsOptional()
  @IsArray()
  @IsString({ each: true })
  @MinLength(3, { each: true })
  tags: string[];

  @ApiPropertyOptional()
  @IsOptional()
  @IsArray()
  @ValidateNested({ each: true })
  @Type(() => CreatPostMetaOptionDto)
  metaOption: CreatPostMetaOptionDto[];
}
